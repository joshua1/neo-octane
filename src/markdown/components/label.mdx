## Usage

```ts
import { Checkbox } from '@/components/ui/checkbox.tsrx'
import { Label } from '@/components/ui/label.tsrx'
```

```tsx

  <div class="flex items-center space-x-2">
    <Checkbox id="terms" />
    <Label for="terms">Accept terms and conditions</Label>
  </div>
</div>
```
