## Usage

Render `<Toaster />` once in your root route (`src/routes/__root.tsrx`).

```ts
import {
  Body,
  createRootRoute,
  Head,
  HeadContent,
  Html,
  Scripts,
} from '@octanejs/tanstack-router'
import type { OctaneNode } from 'octane'

import { Toaster } from '@/components/ui/toast.tsrx'
```

```tsx
export const Route = createRootRoute({
  shellComponent: RootDocument,
})

function RootDocument({ children }: { children?: OctaneNode }) @{
  <Html lang="en">
    <Head>
      <HeadContent />
    </Head>
    <Body class="isolate">
      {children}
      <Toaster />
      <Scripts />
    </Body>
  </Html>
}
```

Then call `toast.add` from anywhere, even outside components.

```ts
import { toast } from '@/components/ui/toast.tsrx'
```

```ts
toast.add({
  title: 'Event created',
  description: 'Sunday, December 3 at 9:00 AM',
})
```

Set `type` to `success`, `info`, `warning`, `error` or `loading` to show an icon, pass `actionProps` to render an action button, or use `toast.promise` to track an async task.

```ts
toast.add({
  type: 'success',
  title: 'Changes saved',
  actionProps: {
    children: 'Undo',
    onClick: () => undo(),
  },
})

toast.promise(saveChanges(), {
  loading: { title: 'Saving...' },
  success: { title: 'Saved' },
  error: { title: 'Could not save' },
})
```

This component replaces <a target="_blank" href="https://sonner.emilkowal.ski">sonner</a>. It is built on the Base UI toast primitive (`@octanejs/base-ui/toast`), so no extra dependency is needed.
