## Usage

```ts
import { Button } from '@/components/ui/button.tsrx'
import {
  ButtonGroup,
  ButtonGroupSeparator,
  ButtonGroupText,
} from '@/components/ui/button-group.tsrx'
```

```tsx
<ButtonGroup>
  <Button variant="neutral">One</Button>
  <Button variant="neutral">Two</Button>
</ButtonGroup>
```
